    CREATE DATABASE LeaveManagementSystem;

-- =============================================
-- PHẦN 1: TẠO CÁC BẢNG
-- =============================================

-- 1. BẢNG PHÒNG BAN
CREATE TABLE Divisions (
    DivisionID INT PRIMARY KEY IDENTITY(1,1),
    DivisionCode VARCHAR(10) UNIQUE NOT NULL,
    DivisionName NVARCHAR(100) NOT NULL,
    Description NVARCHAR(500),
    IsActive BIT DEFAULT 1,
    CreatedAt DATETIME DEFAULT GETDATE(),
    UpdatedAt DATETIME DEFAULT GETDATE()
);

-- 2. BẢNG VAI TRÒ/QUYỀN HẠN
CREATE TABLE Roles (
    RoleID INT PRIMARY KEY IDENTITY(1,1),
    RoleCode VARCHAR(50) UNIQUE NOT NULL,
    RoleName NVARCHAR(100) NOT NULL,
    Description NVARCHAR(500),
    Level INT NOT NULL,
    IsActive BIT DEFAULT 1,
    CreatedAt DATETIME DEFAULT GETDATE(),
    UpdatedAt DATETIME DEFAULT GETDATE()
);

-- 3. BẢNG TÍNH NĂNG HỆ THỐNG
CREATE TABLE Features (
    FeatureID INT PRIMARY KEY IDENTITY(1,1),
    FeatureCode VARCHAR(100) UNIQUE NOT NULL,
    FeatureName NVARCHAR(100) NOT NULL,
    FeaturePath VARCHAR(200),
    Description NVARCHAR(500),
    IsActive BIT DEFAULT 1,
    CreatedAt DATETIME DEFAULT GETDATE()
);

-- 4. BẢNG PHÂN QUYỀN ROLE-FEATURE
CREATE TABLE RoleFeatures (
    RoleFeatureID INT PRIMARY KEY IDENTITY(1,1),
    RoleID INT NOT NULL,
    FeatureID INT NOT NULL,
    CanCreate BIT DEFAULT 0,
    CanRead BIT DEFAULT 0,
    CanUpdate BIT DEFAULT 0,
    CanDelete BIT DEFAULT 0,
    CreatedAt DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (RoleID) REFERENCES Roles(RoleID),
    FOREIGN KEY (FeatureID) REFERENCES Features(FeatureID),
    UNIQUE(RoleID, FeatureID)
);

-- 5. BẢNG NHÂN VIÊN
CREATE TABLE Employees (
    EmployeeID INT PRIMARY KEY IDENTITY(1,1),
    EmployeeCode VARCHAR(20) UNIQUE NOT NULL,
    Username VARCHAR(100) UNIQUE NOT NULL,
    PasswordHash VARCHAR(255) NOT NULL,
    Email VARCHAR(100) UNIQUE NOT NULL,
    FullName NVARCHAR(200) NOT NULL,
    PhoneNumber VARCHAR(20),
    DateOfBirth DATE,
    Gender NVARCHAR(10),
    HireDate DATE NOT NULL,
    DivisionID INT NOT NULL,
    ManagerID INT NULL,
    IsActive BIT DEFAULT 1,
    LastLogin DATETIME,
    CreatedAt DATETIME DEFAULT GETDATE(),
    UpdatedAt DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (DivisionID) REFERENCES Divisions(DivisionID),
    FOREIGN KEY (ManagerID) REFERENCES Employees(EmployeeID)
);

-- 6. BẢNG PHÂN QUYỀN NHÂN VIÊN
CREATE TABLE EmployeeRoles (
    EmployeeRoleID INT PRIMARY KEY IDENTITY(1,1),
    EmployeeID INT NOT NULL,
    RoleID INT NOT NULL,
    AssignedDate DATETIME DEFAULT GETDATE(),
    IsActive BIT DEFAULT 1,
    FOREIGN KEY (EmployeeID) REFERENCES Employees(EmployeeID),
    FOREIGN KEY (RoleID) REFERENCES Roles(RoleID),
    UNIQUE(EmployeeID, RoleID)
);

-- 7. BẢNG LOẠI NGHỈ PHÉP
CREATE TABLE LeaveTypes (
    LeaveTypeID INT PRIMARY KEY IDENTITY(1,1),
    LeaveTypeCode VARCHAR(20) UNIQUE NOT NULL,
    LeaveTypeName NVARCHAR(100) NOT NULL,
    Description NVARCHAR(500),
    DefaultDaysPerYear INT DEFAULT 12,
    RequiresApproval BIT DEFAULT 1,
    IsPaid BIT DEFAULT 1,
    AllowCustomReason BIT DEFAULT 1,
    IsActive BIT DEFAULT 1,
    CreatedAt DATETIME DEFAULT GETDATE()
);

-- 7B. BẢNG LÝ DO NGHỈ PHÉP MẪU
CREATE TABLE LeaveReasonTemplates (
    TemplateID INT PRIMARY KEY IDENTITY(1,1),
    LeaveTypeID INT NOT NULL,
    ReasonText NVARCHAR(500) NOT NULL,
    Description NVARCHAR(500),
    IsActive BIT DEFAULT 1,
    UsageCount INT DEFAULT 0,
    CreatedAt DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (LeaveTypeID) REFERENCES LeaveTypes(LeaveTypeID)
);

-- 8. BẢNG QUOTA NGHỈ PHÉP
CREATE TABLE LeaveQuotas (
    QuotaID INT PRIMARY KEY IDENTITY(1,1),
    EmployeeID INT NOT NULL,
    LeaveTypeID INT NOT NULL,
    Year INT NOT NULL,
    TotalDays DECIMAL(5,1) NOT NULL,
    UsedDays DECIMAL(5,1) DEFAULT 0,
    RemainingDays DECIMAL(5,1) NOT NULL,
    CarryOverDays DECIMAL(5,1) DEFAULT 0,
    CreatedAt DATETIME DEFAULT GETDATE(),
    UpdatedAt DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (EmployeeID) REFERENCES Employees(EmployeeID),
    FOREIGN KEY (LeaveTypeID) REFERENCES LeaveTypes(LeaveTypeID),
    UNIQUE(EmployeeID, LeaveTypeID, Year)
);

-- 9. BẢNG ĐƠN XIN NGHỈ PHÉP
CREATE TABLE LeaveRequests (
    RequestID INT PRIMARY KEY IDENTITY(1,1),
    RequestCode VARCHAR(50) UNIQUE NOT NULL,
    EmployeeID INT NOT NULL,
    LeaveTypeID INT NOT NULL,
    StartDate DATE NOT NULL,
    EndDate DATE NOT NULL,
    TotalDays DECIMAL(5,1) NOT NULL,
    ReasonTemplateID INT NULL,
    CustomReason NVARCHAR(1000) NULL,
    Reason NVARCHAR(1000) NOT NULL,
    Status VARCHAR(20) NOT NULL DEFAULT 'InProgress',
    ProcessedBy INT NULL,
    ProcessedDate DATETIME NULL,
    ProcessedNote NVARCHAR(1000),
    AttachmentPath VARCHAR(500),
    CreatedAt DATETIME DEFAULT GETDATE(),
    UpdatedAt DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (EmployeeID) REFERENCES Employees(EmployeeID),
    FOREIGN KEY (LeaveTypeID) REFERENCES LeaveTypes(LeaveTypeID),
    FOREIGN KEY (ReasonTemplateID) REFERENCES LeaveReasonTemplates(TemplateID),
    FOREIGN KEY (ProcessedBy) REFERENCES Employees(EmployeeID),
    CHECK (EndDate >= StartDate),
    CHECK (Status IN ('InProgress', 'Approved', 'Rejected', 'Cancelled')),
    CHECK (ReasonTemplateID IS NOT NULL OR CustomReason IS NOT NULL)
);

-- 10. BẢNG LỊCH SỬ TRẠNG THÁI ĐƠN
CREATE TABLE RequestStatusHistory (
    HistoryID INT PRIMARY KEY IDENTITY(1,1),
    RequestID INT NOT NULL,
    FromStatus VARCHAR(20),
    ToStatus VARCHAR(20) NOT NULL,
    ChangedBy INT NOT NULL,
    ChangedDate DATETIME DEFAULT GETDATE(),
    Note NVARCHAR(1000),
    FOREIGN KEY (RequestID) REFERENCES LeaveRequests(RequestID),
    FOREIGN KEY (ChangedBy) REFERENCES Employees(EmployeeID)
);

-- 11. BẢNG AUDIT LOGS
CREATE TABLE AuditLogs (
    AuditID BIGINT PRIMARY KEY IDENTITY(1,1),
    TableName VARCHAR(100) NOT NULL,
    RecordID INT NOT NULL,
    Action VARCHAR(20) NOT NULL,
    EmployeeID INT NULL,
    OldValue NVARCHAR(MAX),
    NewValue NVARCHAR(MAX),
    IPAddress VARCHAR(50),
    UserAgent NVARCHAR(500),
    CreatedAt DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (EmployeeID) REFERENCES Employees(EmployeeID)
);

-- 12. BẢNG NGÀY LỄ/NGÀY NGHỈ
CREATE TABLE Holidays (
    HolidayID INT PRIMARY KEY IDENTITY(1,1),
    HolidayName NVARCHAR(200) NOT NULL,
    HolidayDate DATE NOT NULL,
    Year INT NOT NULL,
    IsRecurring BIT DEFAULT 0,
    Description NVARCHAR(500),
    CreatedAt DATETIME DEFAULT GETDATE()
);

-- 13. BẢNG CẤU HÌNH HỆ THỐNG
CREATE TABLE SystemSettings (
    SettingID INT PRIMARY KEY IDENTITY(1,1),
    SettingKey VARCHAR(100) UNIQUE NOT NULL,
    SettingValue NVARCHAR(1000),
    Description NVARCHAR(500),
    UpdatedAt DATETIME DEFAULT GETDATE()
);

-- 14. BẢNG THÔNG BÁO
CREATE TABLE Notifications (
    NotificationID INT PRIMARY KEY IDENTITY(1,1),
    EmployeeID INT NOT NULL,
    Title NVARCHAR(200) NOT NULL,
    Message NVARCHAR(1000) NOT NULL,
    NotificationType VARCHAR(50),
    RelatedRequestID INT NULL,
    IsRead BIT DEFAULT 0,
    CreatedAt DATETIME DEFAULT GETDATE(),
    ReadAt DATETIME NULL,
    FOREIGN KEY (EmployeeID) REFERENCES Employees(EmployeeID),
    FOREIGN KEY (RelatedRequestID) REFERENCES LeaveRequests(RequestID)
);

-- SP1: Tạo đơn nghỉ phép (hỗ trợ custom reason)
GO
CREATE PROCEDURE sp_CreateLeaveRequest
    @EmployeeID INT,
    @LeaveTypeID INT,
    @StartDate DATE,
    @EndDate DATE,
    @ReasonTemplateID INT = NULL,
    @CustomReason NVARCHAR(1000) = NULL,
    @AttachmentPath VARCHAR(500) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        BEGIN TRANSACTION;
        
        -- Kiểm tra phải có ít nhất 1 loại lý do
        IF @ReasonTemplateID IS NULL AND @CustomReason IS NULL
        BEGIN
            RAISERROR(N'Phải cung cấp lý do nghỉ phép (template hoặc custom reason)', 16, 1);
            RETURN;
        END
        
        -- Kiểm tra AllowCustomReason
        DECLARE @AllowCustomReason BIT;
        SELECT @AllowCustomReason = AllowCustomReason 
        FROM LeaveTypes 
        WHERE LeaveTypeID = @LeaveTypeID;
        
        IF @CustomReason IS NOT NULL AND @AllowCustomReason = 0
        BEGIN
            RAISERROR(N'Loại nghỉ phép này không cho phép nhập lý do tùy chỉnh', 16, 1);
            RETURN;
        END
        
        DECLARE @TotalDays DECIMAL(5,1);
        DECLARE @RequestCode VARCHAR(50);
        DECLARE @NewRequestID INT;
        DECLARE @FinalReason NVARCHAR(1000);
        
        -- Tính số ngày làm việc
        SET @TotalDays = (
            SELECT COUNT(*)
            FROM (
                SELECT DATEADD(DAY, number, @StartDate) AS WorkDate
                FROM master..spt_values
                WHERE type = 'P' AND number <= DATEDIFF(DAY, @StartDate, @EndDate)
            ) dates
            WHERE DATEPART(WEEKDAY, WorkDate) NOT IN (1, 7)
        );
        
        -- Xác định lý do cuối cùng
        IF @ReasonTemplateID IS NOT NULL
        BEGIN
            DECLARE @TemplateText NVARCHAR(500);
            SELECT @TemplateText = ReasonText 
            FROM LeaveReasonTemplates 
            WHERE TemplateID = @ReasonTemplateID;
            
            IF @CustomReason IS NOT NULL AND LEN(LTRIM(RTRIM(@CustomReason))) > 0
                SET @FinalReason = @TemplateText + N' ' + @CustomReason;
            ELSE
                SET @FinalReason = @TemplateText;
                
            UPDATE LeaveReasonTemplates 
            SET UsageCount = UsageCount + 1 
            WHERE TemplateID = @ReasonTemplateID;
        END
        ELSE
        BEGIN
            SET @FinalReason = @CustomReason;
        END
        
        -- Tạo mã đơn
        DECLARE @Year VARCHAR(4) = CAST(YEAR(GETDATE()) AS VARCHAR(4));
        DECLARE @SeqNum VARCHAR(4);
        
        SELECT @SeqNum = RIGHT('0000' + CAST(COUNT(*) + 1 AS VARCHAR), 4)
        FROM LeaveRequests
        WHERE YEAR(CreatedAt) = YEAR(GETDATE());
        
        SET @RequestCode = 'LR-' + @Year + '-' + @SeqNum;
        
        -- Insert đơn nghỉ phép
        INSERT INTO LeaveRequests (
            RequestCode, EmployeeID, LeaveTypeID, StartDate, EndDate, 
            TotalDays, ReasonTemplateID, CustomReason, Reason, Status, AttachmentPath
        )
        VALUES (
            @RequestCode, @EmployeeID, @LeaveTypeID, @StartDate, @EndDate,
            @TotalDays, @ReasonTemplateID, @CustomReason, @FinalReason, 'InProgress', @AttachmentPath
        );
        
        SET @NewRequestID = SCOPE_IDENTITY();
        
        -- Ghi lịch sử trạng thái
        INSERT INTO RequestStatusHistory (RequestID, FromStatus, ToStatus, ChangedBy, Note)
        VALUES (@NewRequestID, NULL, 'InProgress', @EmployeeID, N'Tạo đơn mới');
        
        -- Ghi audit log
        INSERT INTO AuditLogs (TableName, RecordID, Action, EmployeeID, NewValue)
        VALUES ('LeaveRequests', @NewRequestID, 'INSERT', @EmployeeID, 
                CONCAT('RequestCode:', @RequestCode, ', StartDate:', @StartDate, ', EndDate:', @EndDate, ', Reason:', @FinalReason));
        
        -- Tạo thông báo cho quản lý
        DECLARE @ManagerID INT;
        SELECT @ManagerID = ManagerID FROM Employees WHERE EmployeeID = @EmployeeID;
        
        IF @ManagerID IS NOT NULL
        BEGIN
            INSERT INTO Notifications (EmployeeID, Title, Message, NotificationType, RelatedRequestID)
            VALUES (@ManagerID, N'Đơn nghỉ phép mới cần xét duyệt', 
                    CONCAT(N'Bạn có đơn nghỉ phép mới từ nhân viên cần xét duyệt. Mã đơn: ', @RequestCode),
                    'LeaveRequest', @NewRequestID);
        END
        
        COMMIT TRANSACTION;
        
        SELECT @NewRequestID AS RequestID, @RequestCode AS RequestCode, @FinalReason AS FinalReason, 'SUCCESS' AS Status;
        
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();
        RAISERROR(@ErrorMessage, 16, 1);
    END CATCH
END;
GO

-- SP2: Xét duyệt đơn nghỉ phép
GO
CREATE PROCEDURE sp_ProcessLeaveRequest
    @RequestID INT,
    @ProcessedBy INT,
    @Action VARCHAR(20),
    @ProcessedNote NVARCHAR(1000) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        BEGIN TRANSACTION;
        
        DECLARE @NewStatus VARCHAR(20);
        DECLARE @OldStatus VARCHAR(20);
        DECLARE @EmployeeID INT;
        DECLARE @LeaveTypeID INT;
        DECLARE @TotalDays DECIMAL(5,1);
        DECLARE @Year INT = YEAR(GETDATE());
        
        SELECT 
            @OldStatus = Status,
            @EmployeeID = EmployeeID,
            @LeaveTypeID = LeaveTypeID,
            @TotalDays = TotalDays
        FROM LeaveRequests
        WHERE RequestID = @RequestID;
        
        IF @OldStatus != 'InProgress'
        BEGIN
            RAISERROR(N'Đơn này đã được xử lý rồi!', 16, 1);
            RETURN;
        END
        
        SET @NewStatus = CASE WHEN @Action = 'APPROVE' THEN 'Approved' ELSE 'Rejected' END;
        
        UPDATE LeaveRequests
        SET Status = @NewStatus,
            ProcessedBy = @ProcessedBy,
            ProcessedDate = GETDATE(),
            ProcessedNote = @ProcessedNote,
            UpdatedAt = GETDATE()
        WHERE RequestID = @RequestID;
        
        INSERT INTO RequestStatusHistory (RequestID, FromStatus, ToStatus, ChangedBy, Note)
        VALUES (@RequestID, @OldStatus, @NewStatus, @ProcessedBy, @ProcessedNote);
        
        IF @Action = 'APPROVE'
        BEGIN
            UPDATE LeaveQuotas
            SET UsedDays = UsedDays + @TotalDays,
                RemainingDays = RemainingDays - @TotalDays,
                UpdatedAt = GETDATE()
            WHERE EmployeeID = @EmployeeID 
                AND LeaveTypeID = @LeaveTypeID 
                AND Year = @Year;
        END
        
        INSERT INTO AuditLogs (TableName, RecordID, Action, EmployeeID, OldValue, NewValue)
        VALUES ('LeaveRequests', @RequestID, 'UPDATE', @ProcessedBy,
                CONCAT('Status:', @OldStatus),
                CONCAT('Status:', @NewStatus, ', ProcessedBy:', @ProcessedBy));
        
        DECLARE @NotifMessage NVARCHAR(1000);
        SET @NotifMessage = CASE 
            WHEN @Action = 'APPROVE' THEN N'Đơn nghỉ phép của bạn đã được phê duyệt'
            ELSE N'Đơn nghỉ phép của bạn đã bị từ chối'
        END;
        
        INSERT INTO Notifications (EmployeeID, Title, Message, NotificationType, RelatedRequestID)
        VALUES (@EmployeeID, @NotifMessage, 
                ISNULL(@ProcessedNote, N'Không có ghi chú'),
                CASE WHEN @Action = 'APPROVE' THEN 'Approval' ELSE 'Rejection' END,
                @RequestID);
        
        COMMIT TRANSACTION;
        
        SELECT 'SUCCESS' AS Status, @NewStatus AS NewStatus;
        
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();
        RAISERROR(@ErrorMessage, 16, 1);
    END CATCH
END;
GO

-- SP3: Lấy danh sách đơn theo phân quyền
GO
CREATE PROCEDURE sp_GetLeaveRequests
    @EmployeeID INT,
    @Status VARCHAR(20) = NULL,
    @StartDate DATE = NULL,
    @EndDate DATE = NULL
AS
BEGIN
    SET NOCOUNT ON;
    
    DECLARE @RoleLevel INT;
    DECLARE @DivisionID INT;
    
    SELECT TOP 1 
        @RoleLevel = r.Level,
        @DivisionID = e.DivisionID
    FROM Employees e
    INNER JOIN EmployeeRoles er ON e.EmployeeID = er.EmployeeID
    INNER JOIN Roles r ON er.RoleID = r.RoleID
    WHERE e.EmployeeID = @EmployeeID
    ORDER BY r.Level;
    
    SELECT 
        lr.RequestID,
        lr.RequestCode,
        lr.StartDate,
        lr.EndDate,
        lr.TotalDays,
        lr.Reason,
        lr.Status,
        lr.CreatedAt,
        e.EmployeeCode,
        e.FullName AS EmployeeName,
        d.DivisionName,
        lt.LeaveTypeName,
        p.FullName AS ProcessedByName,
        lr.ProcessedDate,
        lr.ProcessedNote
    FROM LeaveRequests lr
    INNER JOIN Employees e ON lr.EmployeeID = e.EmployeeID
    INNER JOIN Divisions d ON e.DivisionID = d.DivisionID
    INNER JOIN LeaveTypes lt ON lr.LeaveTypeID = lt.LeaveTypeID
    LEFT JOIN Employees p ON lr.ProcessedBy = p.EmployeeID
    WHERE 
        (
            @RoleLevel <= 1
            OR
            (@RoleLevel = 2 AND e.DivisionID = @DivisionID)
            OR
            (@RoleLevel = 3 AND (e.ManagerID = @EmployeeID OR e.EmployeeID = @EmployeeID))
            OR
            (@RoleLevel >= 4 AND e.EmployeeID = @EmployeeID)
        )
        AND (@Status IS NULL OR lr.Status = @Status)
        AND (@StartDate IS NULL OR lr.StartDate >= @StartDate)
        AND (@EndDate IS NULL OR lr.EndDate <= @EndDate)
    ORDER BY lr.CreatedAt DESC;
END;
GO

-- SP4: Lấy agenda nghỉ phép của phòng ban
GO
CREATE PROCEDURE sp_GetDivisionAgenda
    @DivisionID INT,
    @StartDate DATE,
    @EndDate DATE
AS
BEGIN
    SET NOCOUNT ON;
    
    SELECT 
        e.EmployeeID,
        e.EmployeeCode,
        e.FullName,
        d.DateValue,
        CASE 
            WHEN lr.RequestID IS NOT NULL THEN 'OnLeave'
            WHEN DATEPART(WEEKDAY, d.DateValue) IN (1, 7) THEN 'Weekend'
            WHEN h.HolidayID IS NOT NULL THEN 'Holiday'
            ELSE 'Working'
        END AS Status,
        lt.LeaveTypeName,
        lr.RequestCode
    FROM Employees e
    CROSS JOIN (
        SELECT DATEADD(DAY, number, @StartDate) AS DateValue
        FROM master..spt_values
        WHERE type = 'P' 
            AND DATEADD(DAY, number, @StartDate) <= @EndDate
    ) d
    LEFT JOIN LeaveRequests lr ON 
        e.EmployeeID = lr.EmployeeID
        AND lr.Status = 'Approved'
        AND d.DateValue BETWEEN lr.StartDate AND lr.EndDate
    LEFT JOIN LeaveTypes lt ON lr.LeaveTypeID = lt.LeaveTypeID
    LEFT JOIN Holidays h ON d.DateValue = h.HolidayDate
    WHERE 
        e.DivisionID = @DivisionID
        AND e.IsActive = 1
    ORDER BY e.EmployeeCode, d.DateValue;
END;
GO

-- SP6: Báo cáo thống kê nghỉ phép
GO
CREATE PROCEDURE sp_GetLeaveStatistics
    @DivisionID INT = NULL,
    @Year INT = NULL,
    @Month INT = NULL
AS
BEGIN
    SET NOCOUNT ON;
    
    IF @Year IS NULL
        SET @Year = YEAR(GETDATE());
    
    SELECT 
        d.DivisionName,
        lt.LeaveTypeName,
        COUNT(lr.RequestID) AS TotalRequests,
        SUM(CASE WHEN lr.Status = 'Approved' THEN 1 ELSE 0 END) AS ApprovedRequests,
        SUM(CASE WHEN lr.Status = 'Rejected' THEN 1 ELSE 0 END) AS RejectedRequests,
        SUM(CASE WHEN lr.Status = 'InProgress' THEN 1 ELSE 0 END) AS PendingRequests,
        SUM(CASE WHEN lr.Status = 'Approved' THEN lr.TotalDays ELSE 0 END) AS TotalLeaveDays,
        AVG(CASE WHEN lr.Status = 'Approved' THEN lr.TotalDays ELSE NULL END) AS AvgLeaveDays
    FROM LeaveRequests lr
    INNER JOIN Employees e ON lr.EmployeeID = e.EmployeeID
    INNER JOIN Divisions d ON e.DivisionID = d.DivisionID
    INNER JOIN LeaveTypes lt ON lr.LeaveTypeID = lt.LeaveTypeID
    WHERE 
        YEAR(lr.CreatedAt) = @Year
        AND (@Month IS NULL OR MONTH(lr.CreatedAt) = @Month)
        AND (@DivisionID IS NULL OR e.DivisionID = @DivisionID)
    GROUP BY d.DivisionName, lt.LeaveTypeName
    ORDER BY d.DivisionName, lt.LeaveTypeName;
END;
GO
